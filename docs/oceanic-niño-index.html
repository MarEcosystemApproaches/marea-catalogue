10.2 View Data 

library(plotly)
plotly_df <- data@data %>%
  mutate(date = lubridate::make_date(year, month, day = 1),
         smooth_1yr = zoo::rollapply(value_value, mean, width = 12, partial = TRUE),
         smooth_10yr = zoo::rollapply(value_value, mean, width = 120, partial = TRUE),
         overall_mean = mean(value_value))


p <- plot_ly(plotly_df, x = ~date) %>%
  add_lines(y = ~value_value,
            name = "Monthly Anomaly",
            line = list(color = "lightgrey"),
            hovertemplate = "Anomaly Value: %{y:.2f} <extra></extra>"
  ) %>%
  add_lines(y = ~smooth_1yr,
            name = "1-yr Smooth",
            line = list(color = "blue", width = 2),
            hovertemplate = "1-yr Smooth: %{y:.2f} <extra></extra>"
  ) %>%
  add_lines(y = ~smooth_10yr,
            name = "10-yr Smooth",
            line = list(color = "red", width = 2),
            hovertemplate = "10-yr Smooth: %{y:.2f} <extra></extra>") %>%
  add_lines(y = ~overall_mean,
            name = "Overall Mean",
            line = list(color = "black", width = 2, dash = "dash"),
            hovertemplate = "Timeseries Mean: %{y:.2f} <extra></extra>") %>%
  layout(
    title = "North Pacific Gyre Oscillation Index",
    xaxis = list(title = "Date",
                 hoverformat = "%b %Y" ),
    yaxis = list(title = "NPGO Index (normalized anomaly)",
                 fixedrange = T),
    hovermode = "x unified",
    margin = list(t = 80)
    
  ) %>%
  config(displayModeBar = FALSE)

p
